cmake_minimum_required(VERSION 3.14)
project(TIVIDY_Engine)

set(CMAKE_CXX_STANDARD 11)

# Bring in GoogleTest
include(FetchContent)
FetchContent_Declare(
  googletest
  URL https://github.com/google/googletest/archive/03597a01ee50ed33e9dfd640b249b4be3799d395.zip
)
FetchContent_MakeAvailable(googletest)

# Define your source files (excluding any main.cpp you might use for the TUI later)
set(CORE_SOURCES 
    src/WorkItem.cpp 
    src/CreatedState.cpp 
    src/AvailableState.cpp 
    src/AssignedState.cpp 
    src/StartedState.cpp 
    src/CompletedState.cpp
    src/RejectedState.cpp
    src/CancelledState.cpp
    src/EscalatedState.cpp
    src/WorkflowInstance.cpp
    src/WorkflowDefinition.cpp
    src/WorkflowEngine.cpp
)

# Create the test executable
add_executable(EngineTests tests/EngineTests.cpp ${CORE_SOURCES})

# Link GoogleTest
target_link_libraries(EngineTests gtest_main)

include(GoogleTest)
gtest_discover_tests(EngineTests)
